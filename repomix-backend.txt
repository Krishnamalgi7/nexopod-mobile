import {

import DateTimePicker from "@react-native-community/datetimepicker";
import { ArrowLeft, CalendarDays, Clock, Store } from "lucide-react-native";
import { API_BASE_URL } from "@/config/api";
import * as SecureStore from "expo-secure-store";

type StoreService = "Phone Charging" | "Pick Me Later";

interface DayTiming {
  isClosed: boolean;
  openTime?: string;
  closeTime?: string;
}

interface StoreData {
  _id: string;
  name: string;
  services: StoreService[];
  weeklyTimings?: Record<string, DayTiming>;
}

interface BookedRange {
  startAt: string;
  endAt: string;
}

export default function CreateBookingScreen() {
  const { storeId, service } = useLocalSearchParams<{
    storeId: string;
    service: StoreService;
  }>();

  const [store, setStore] = useState<StoreData | null>(null);
  const [loading, setLoading] = useState(true);

  const [selectedDate, setSelectedDate] = useState(new Date());
  const [startTime, setStartTime] = useState(new Date());
  const [endTime, setEndTime] = useState(new Date());

  const [showDatePicker, setShowDatePicker] = useState(false);
  const [showStartPicker, setShowStartPicker] = useState(false);
  const [showEndPicker, setShowEndPicker] = useState(false);

  const [submitting, setSubmitting] = useState(false);

  const [bookedTimes, setBookedTimes] = useState<string[]>([]);
  const [bookedRanges, setBookedRanges] = useState<BookedRange[]>([]);
  const [availabilityLoading, setAvailabilityLoading] = useState(false);
  const [availabilityError, setAvailabilityError] = useState("");

  const fetchStore = useCallback(async () => {
    try {
      const response = await fetch(
        `${API_BASE_URL}/api/public/stores/${storeId}`,
      );

      const result = await response.json();

      if (!response.ok || !result.success) {
        throw new Error(result.message || "Unable to load store details.");
      }

      setStore(result.data);
    } catch (error) {
      Alert.alert(
        "Unable to load store",
        error instanceof Error ? error.message : "Please try again.",
      );
    } finally {
      setLoading(false);
    }
  }, [storeId]);

  useEffect(() => {
    if (storeId) {
      fetchStore();
    } else {
      setLoading(false);
    }
  }, [storeId, fetchStore]);

  useEffect(() => {
    const fetchAvailability = async () => {
      if (!storeId || !service) return;

      setAvailabilityLoading(true);
      setAvailabilityError("");
      setBookedTimes([]);
      setBookedRanges([]);

      try {
        const endpoint =
          service === "Phone Charging" ? "phone-charging" : "pick-me-later";

        const response = await fetch(
          `${API_BASE_URL}/api/public/stores/${storeId}/${endpoint}/availability?date=${formatDate(selectedDate)}`,
        );

        const result = await response.json();

        if (!response.ok || !result.success) {
          throw new Error(result.message || "Unable to fetch availability.");
        }

        if (service === "Phone Charging") {
          setBookedTimes(result.data.bookedTimes || []);
        } else {
          setBookedRanges(result.data.bookedRanges || []);
        }
      } catch (error) {
        setAvailabilityError(
          error instanceof Error
            ? error.message
            : "Unable to fetch availability.",
        );
      } finally {
        setAvailabilityLoading(false);
      }
    };

    fetchAvailability();
  }, [storeId, service, selectedDate]);

  const formatDate = (date: Date) =>
    `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}-${String(date.getDate()).padStart(2, "0")}`;

  const handleSubmit = async () => {
    if (availabilityLoading || availabilityError) {
      Alert.alert("Please wait", "Availability could not be verified.");
      return;
    }

    const startAt = new Date(
      selectedDate.getFullYear(),
      selectedDate.getMonth(),

  Image,
  type ImageProps,
  type ImageSourcePropType,
} from "react-native";

interface RemoteImageProps extends Omit<ImageProps, "source"> {
  uri?: string | null;
}

const FALLBACK_IMAGE =
  require("../../../assets/images/user_page/fallback.png");

export default function RemoteImage({
  uri,
  onError,
  ...props
}: RemoteImageProps) {
  const [hasError, setHasError] = useState(false);

  useEffect(() => {
    setHasError(false);
  }, [uri]);

  const source: ImageSourcePropType =
    !uri || hasError ? FALLBACK_IMAGE : { uri };

  return (
    <Image
      {...props}
      source={source}
      onError={(event) => {
        setHasError(true);
        onError?.(event);
      }}
    />
  );
}

  const [error, setError] = useState("");

  const [otpRequested, setOtpRequested] = useState(false);
  const [secondsRemaining, setSecondsRemaining] = useState(0);

  // 1. Request a new OTP from the backend.
  const handleSendOtp = async () => {
    if (!/^[0-9]{10}$/.test(mobileNumber)) {
      setError("Enter a valid 10-digit mobile number.");
      return;
    }

    setLoading(true);
    setError("");

    try {
      // 2. Request OTP for the registered store number.
      const response = await fetch(`${API_BASE_URL}/api/role-auth/send-otp`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ phone: mobileNumber }),
      });

      const result = await response.json();

      if (!response.ok || !result.success) {
        setError(result.message || "Unable to send OTP.");
        return;
      }

      // 3. Start a fresh two-minute OTP session.
      setOtpRequested(true);
      setOtp("");
      setSecondsRemaining(result.data.expiresIn || 120);

      console.log("[ShopOwner] OTP request successful.");
    } catch (error) {
      // 4. Display the actual network error for debugging.
      const errorMessage =
        error instanceof Error ? error.message : String(error);

      console.error("[ShopOwner] OTP request failed:", errorMessage);

      setError(`Connection failed: ${errorMessage}`);
    } finally {
      setLoading(false);
    }
  };

  // 4. Update the OTP countdown every second.
  useEffect(() => {
    if (!otpRequested || secondsRemaining <= 0) return;

    const timer = setInterval(() => {
      setSecondsRemaining((previous) => Math.max(previous - 1, 0));
    }, 1000);

    return () => clearInterval(timer);
  }, [otpRequested, secondsRemaining]);

  // 5. Format the remaining OTP validity time.
  const formattedTime = `${Math.floor(secondsRemaining / 60)
    .toString()
    .padStart(2, "0")}:${(secondsRemaining % 60).toString().padStart(2, "0")}`;

  // 1. Validate the login inputs.
  const handleLogin = async () => {
    // 1. Ensure an OTP has been requested and is still valid.
    if (!otpRequested || secondsRemaining <= 0) {
      setError("OTP expired. Please request a new OTP.");
      return;
    }

    if (!/^[0-9]{10}$/.test(mobileNumber)) {
      setError("Enter a valid 10-digit mobile number.");
      return;
    }

    if (!/^[0-9]{4}$/.test(otp)) {
      setError("Enter a valid 4-digit OTP.");
      return;

    // 2. Start the login request.
    setLoading(true);
    setError("");

    try {
      // 3. Send credentials to the Shop Owner login API.
      const response = await fetch(`${API_BASE_URL}/api/role-auth/verify-otp`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          phone: mobileNumber,
          otp,
        }),
      });

      // 4. Read the API response.
      const result = await response.json();

      if (!response.ok || !result.success) {
        setError(result.message || "Login failed. Please try again.");
        return;
      }

      // 5. Securely save the authentication token.
      // Handle users who have both roles.

          params: {
            selectionToken: result.data.selectionToken,
          },
        });
        return;

        router.replace(
          result.data.customer.name?.trim()
            ? "/user/home"
            : "/user/complete-profile",
        );
        return;

          result.data.store.isActive
            ? "/shopowner/dashboard"
            : "/shopowner/profile",
        );
        return;
      }

      setError("Unable to identify your account role. Please try again.");
    } catch (error) {
      console.error(
        "[ShopOwner] Login request failed:",
        error instanceof Error ? error.message : "Unknown error",
      );

      setError("Unable to connect to the server. Please try again.");
    } finally {
      // 7. Stop the loading state.
      setLoading(false);

  return (

              Welcome back! Sign in to manage your store.
            </Text>
          </View>

          {/* 3. Display the login form panel. */}
          <View className="rounded-[28px] border border-[#DCEFE8] bg-white p-5 shadow-lg">
            <Text className="mb-2 text-sm font-semibold text-[#244B5A]">

import { useCallback, useEffect, useRef, useState } from "react";
import { ActivityIndicator, Pressable, Text, View } from "react-native";

  connectShopOwnerSocket,

interface Booking {
  _id: string;
  service: "Phone Charging" | "Pick Me Later";
  startAt: string;
  status: "confirmed" | "in_progress" | "completed";
  customerId?: {
    _id: string;
    name: string;
  } | null;
  nextAction: {
    label: string;
    status: "in_progress" | "completed";
  } | null;
}

export default function BookingsScreen() {
  const router = useRouter();

  const [phoneChargingBookings, setPhoneChargingBookings] = useState<Booking[]>(
    [],
  );
  const [pickMeLaterBookings, setPickMeLaterBookings] = useState<Booking[]>([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);

  const [error, setError] = useState("");
  const [isStoreActive, setIsStoreActive] = useState<boolean | null>(null);
  const [activationModalVisible, setActivationModalVisible] = useState(false);
  const [selectedService, setSelectedService] = useState<
    "all" | "Phone Charging" | "Pick Me Later"

  const [nextCursor, setNextCursor] = useState<string | null>(null);
  const loadedCursors = useRef<string[]>([]);
  const [hasNextPage, setHasNextPage] = useState(false);
  const [loadingMore, setLoadingMore] = useState(false);

  useEffect(() => {

        const token = await SecureStore.getItemAsync("shopOwnerToken");

        if (!token) {
          router.replace("/auth/login");
          return;
        }

        const response = await fetch(`${API_BASE_URL}/api/shop-owners/me`, {

            Authorization: `Bearer ${token}`,
          },

        const result = await response.json();

        if (response.status === 401) {
          await SecureStore.deleteItemAsync("shopOwnerToken");
          router.replace("/auth/login");
          return;
        }

        if (!response.ok || !result.success) {
          throw new Error(result.message || "Unable to fetch store status.");

        setIsStoreActive(active);

        if (!active) {
          setActivationModalVisible(true);

          error instanceof Error

            : "Unable to fetch store status.",
        );
      }

    fetchStoreStatus();
  }, [router]);

      cursor: string | null = null,
      append = false,
      rememberCursor = false,
      service = selectedService,

      try {
        setError("");

        const token = await SecureStore.getItemAsync("shopOwnerToken");

        if (!token) {
          router.replace("/auth/login");
          return null;

          });

          return Array.from(uniqueBookings.values());
        });

        setPickMeLaterBookings((previous) => {
          if (!append) return storageBookings;

          const uniqueBookings = new Map(
            previous.map((booking) => [booking._id, booking]),
          );

          storageBookings.forEach((booking: Booking) => {
            uniqueBookings.set(booking._id, booking);
          });

          return Array.from(uniqueBookings.values());
        });

        const responseNextCursor = result.data.pagination?.nextCursor || null;

        setNextCursor(responseNextCursor);
        setHasNextPage(result.data.pagination?.hasNextPage || false);

        if (!append) {
          loadedCursors.current = [];
        } else if (rememberCursor && cursor) {
          loadedCursors.current = [...loadedCursors.current, cursor];
        }

        return responseNextCursor;
      } catch (error) {
        setError(
          error instanceof Error
            ? error.message
            : "Unable to connect to the server.",
        );

        return null;
      } finally {
        setLoading(false);
        setRefreshing(false);
        setLoadingMore(false);
      }
    },
    [router, selectedService],

  // Refresh all currently loaded booking pages
  const refreshLoadedBookings = useCallback(async () => {
    const pagesToReload = loadedCursors.current.length + 1;
    let cursor: string | null = null;

    for (let page = 0; page < pagesToReload; page++) {
      const nextPageCursor = await fetchBookings(cursor, page > 0, page > 0);

      if (!nextPageCursor) break;

      cursor = nextPageCursor;
    }
  }, [fetchBookings]);

  // Set up socket connection and event listeners
  useEffect(() => {
    fetchBookings();

    let isMounted = true;

    const unsubscribe = subscribeToBookingEvents(() => {
      if (isMounted) {
        refreshLoadedBookings();
      }
    });

    connectShopOwnerSocket();

    return () => {
      isMounted = false;
      unsubscribe();
    };
  }, [fetchBookings, refreshLoadedBookings]);

  const handleServiceFilter = (
    service: "all" | "Phone Charging" | "Pick Me Later",
  ) => {
    if (service === selectedService) return;

    setSelectedService(service);
    loadedCursors.current = [];
    setLoading(true);
  };

  // Handle pull-to-refresh
  const handleRefresh = () => {
    setRefreshing(true);
    refreshLoadedBookings();
  };

  // Load the next set of bookings using the cursor.
  const handleLoadMore = () => {
    if (!hasNextPage || loadingMore || !nextCursor) return;

    setLoadingMore(true);
    fetchBookings(nextCursor, true, true);
  };

  // Handle booking status update
  const handleUpdateStatus = async (
    bookingId: string,
    status: "in_progress" | "completed",
  ) => {
    try {
      setUpdatingBookingId(bookingId);
      setError("");

      const token = await SecureStore.getItemAsync("shopOwnerToken");

      if (!token) {
        router.replace("/auth/login");
        return;
      }

import { useCallback, useEffect, useState } from "react";
import {
  ActivityIndicator,
  Alert,
  Pressable,
  Switch,
  Text,
  View,
} from "react-native";
import { useRouter } from "expo-router";
import * as SecureStore from "expo-secure-store";
import { SafeAreaView } from "react-native-safe-area-context";
import { Zap, Package } from "lucide-react-native";

import { API_BASE_URL } from "@/config/api";
import Loader from "@/components/common/Loader";
import RefreshableScrollView from "@/components/common/RefreshableScrollView";

type StoreService = "Phone Charging" | "Pick Me Later";

interface StoreData {
  services: StoreService[];
}

export default function ServicesScreen() {
  const router = useRouter();

  const [services, setServices] = useState<StoreService[]>([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [error, setError] = useState("");
  const [saving, setSaving] = useState(false);
  const [success, setSuccess] = useState("");

  const fetchServices = useCallback(async () => {
    try {
      setError("");

      const token = await SecureStore.getItemAsync("shopOwnerToken");

      if (!token) {
        router.replace("/auth/login");
        return;
      }

      const response = await fetch(`${API_BASE_URL}/api/shop-owners/me`, {
        method: "GET",
        headers: {
          Authorization: `Bearer ${token}`,
          "Content-Type": "application/json",
        },
      });

      const result = await response.json();

      if (response.status === 401 || response.status === 403) {
        await SecureStore.deleteItemAsync("shopOwnerToken");
        router.replace("/auth/login");
        return;
      }

      if (!response.ok) {
        throw new Error(result.message || "Unable to fetch services.");
      }

      const store: StoreData = result.data.store || result.data;
      setServices(store.services || []);
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : "Unable to connect to the server.",
      );
    } finally {
      setLoading(false);
      setRefreshing(false);
    }
  }, [router]);

  useEffect(() => {
    fetchServices();
  }, [fetchServices]);

  const handleRefresh = () => {
    setRefreshing(true);
    fetchServices();
  };

  const handleToggleService = (serviceName: StoreService, enabled: boolean) => {
    const action = enabled ? "enable" : "disable";

    Alert.alert(
      `${enabled ? "Enable" : "Disable"} ${serviceName}?`,
      `Are you sure you want to ${action} this service?`,
      [
        {
          text: "Cancel",
          style: "cancel",
        },
        {
          text: "Confirm",
          onPress: () => {
            setServices((previous) =>
              enabled
                ? [...previous, serviceName]
                : previous.filter((service) => service !== serviceName),
            );

            setError("");
            setSuccess("");
          },
        },
      ],
    );
  };

  const handleSaveServices = async () => {
    try {
      setSaving(true);
      setError("");
      setSuccess("");

      const token = await SecureStore.getItemAsync("shopOwnerToken");

      if (!token) {
        router.replace("/auth/login");
        return;
      }

      const response = await fetch(`${API_BASE_URL}/api/shop-owners/services`, {
        method: "PATCH",
        headers: {
          Authorization: `Bearer ${token}`,
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ services }),
      });

      const result = await response.json();

      if (response.status === 401 || response.status === 403) {
        await SecureStore.deleteItemAsync("shopOwnerToken");
        router.replace("/auth/login");
        return;
      }

      if (!response.ok || !result.success) {
        throw new Error(result.message || "Unable to update services.");
      }

      setServices(result.data.services || []);
      setSuccess("Services updated successfully.");
    } catch (error) {
      setError(
        error instanceof Error ? error.message : "Unable to update services.",
      );
    } finally {
      setSaving(false);
    }
  };

  if (loading) {
    return <Loader />;
  }

  return (
    <SafeAreaView className="flex-1 bg-[#F8FAFC]">
      <RefreshableScrollView
        className="flex-1"

import { useEffect, useState } from "react";
import { useLocalSearchParams, useRouter } from "expo-router";
import * as SecureStore from "expo-secure-store";
import { API_BASE_URL } from "../../config/api";

import {
  KeyboardAvoidingView,
  Platform,
  Pressable,
  ScrollView,
  Text,
  TextInput,
  View,
} from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";
import { LockKeyhole, Smartphone, ShieldCheck } from "lucide-react-native";

export default function UserLoginScreen() {
  const router = useRouter();
  const { storeId, service } = useLocalSearchParams<{
    storeId?: string;
    service?: string;
  }>();
  const [mobileNumber, setMobileNumber] = useState("");
  const [otp, setOtp] = useState("");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");
  const [otpRequested, setOtpRequested] = useState(false);
  const [secondsRemaining, setSecondsRemaining] = useState(0);

  const handleSendOtp = async () => {
    if (!/^[0-9]{10}$/.test(mobileNumber)) {
      setError("Enter a valid 10-digit mobile number.");
      return;
    }

    setLoading(true);
    setError("");

    try {
      const response = await fetch(`${API_BASE_URL}/api/customers/send-otp`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ phone: mobileNumber }),
      });

      const result = await response.json();

      if (!response.ok || !result.success) {
        setError(result.message || "Unable to send OTP.");
        return;
      }

      setOtpRequested(true);
      setOtp("");
      setSecondsRemaining(result.data.expiresIn || 120);
    } catch (error) {
      setError("Unable to connect to the server. Please try again.");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    if (!otpRequested || secondsRemaining <= 0) return;

    const timer = setInterval(() => {
      setSecondsRemaining((previous) => Math.max(previous - 1, 0));
    }, 1000);

    return () => clearInterval(timer);
  }, [otpRequested, secondsRemaining]);

  const formattedTime = `${Math.floor(secondsRemaining / 60)
    .toString()
    .padStart(2, "0")}:${(secondsRemaining % 60).toString().padStart(2, "0")}`;

  const handleLogin = async () => {
    if (!otpRequested || secondsRemaining <= 0) {
      setError("OTP expired. Please request a new OTP.");
      return;
    }

    if (!/^[0-9]{4}$/.test(otp)) {
      setError("Enter a valid 4-digit OTP.");
      return;
    }

    setLoading(true);
    setError("");

    try {
      const response = await fetch(`${API_BASE_URL}/api/customers/login`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          phone: mobileNumber,
          otp,
        }),
      });

      const result = await response.json();

      if (!response.ok || !result.success) {
        setError(result.message || "Login failed. Please try again.");
        return;
      }

      await SecureStore.setItemAsync("customerToken", result.data.token);

      if (storeId && service) {
        router.replace({
          pathname: "/user/store/store-details",
          params: { storeId, service },
        });
      } else {
        router.replace("/user/home");
      }
    } catch (error) {
      setError("Unable to connect to the server. Please try again.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <SafeAreaView className="flex-1 bg-[#F0FFFA]" edges={["top", "bottom"]}>
      <KeyboardAvoidingView
        className="flex-1"
        behavior={Platform.OS === "ios" ? "padding" : "height"}
      >
        <ScrollView
          className="flex-1"
          contentContainerClassName="flex-grow justify-start px-6 pt-10 pb-8"
          keyboardShouldPersistTaps="handled"
          showsVerticalScrollIndicator={false}
        >
          <View className="flex-1">
            <View className="mb-10">
              <View className="h-14 w-14 items-center justify-center rounded-2xl bg-[#D9F8EF]">
                <Smartphone size={28} color="#00BFA5" />
              </View>

              <Text className="mt-8 text-3xl font-bold text-[#142D46]">
                Welcome to NexoPod
              </Text>
              <Text className="mt-3 text-base leading-6 text-[#7B8798]">
                Discover stores, explore services and manage all your bookings
                in one place.
              </Text>
            </View>

            <View className="rounded-3xl border border-[#E7EDF1] bg-white p-5">
              <Text className="text-xl font-bold text-[#142D46]">
                {otpRequested ? "Verify your number" : "Login or Sign Up"}
              </Text>

              <Text className="mt-2 text-sm text-[#7B8798]">
                {otpRequested
                  ? `Enter the OTP sent to +91 ${mobileNumber}`
                  : "Enter your mobile number to continue"}
              </Text>

              <Text className="mb-2 mt-7 text-sm font-semibold text-[#142D46]">
                Mobile Number
              </Text>

              <View className="flex-row items-center rounded-xl border border-[#E7EDF1] bg-[#F8FAFB] px-4">
                <Text className="mr-3 text-base font-semibold text-[#142D46]">
                  +91
                </Text>
                <TextInput
                  className="h-14 flex-1 text-base text-[#142D46]"
                  placeholder="Enter 10-digit number"
                  placeholderTextColor="#A0AAB5"
                  keyboardType="number-pad"
                  maxLength={10}
                  value={mobileNumber}
                  onChangeText={(value) => {
                    setMobileNumber(value.replace(/[^0-9]/g, ""));
                    setError("");
                  }}
                  editable={!otpRequested && !loading}
                />
              </View>

              {otpRequested && (
                <>
                  <Text className="mb-2 mt-6 text-sm font-semibold text-[#142D46]">
                    One-Time Password
                  </Text>

                  <View className="flex-row items-center rounded-xl border border-[#E7EDF1] bg-[#F8FAFB] px-4">
                    <LockKeyhole size={19} color="#00BFA5" />
                    <TextInput
                      className="h-14 flex-1 px-3 text-lg tracking-[8px] text-[#142D46]"
                      placeholder="Enter OTP"
                      placeholderTextColor="#A0AAB5"
                      keyboardType="number-pad"
                      maxLength={4}
                      value={otp}
                      onChangeText={(value) => {
                        setOtp(value.replace(/[^0-9]/g, ""));
                        setError("");
                      }}
                    />
                  </View>

                  <View className="mt-3 flex-row items-center justify-between">
                    <Text className="text-xs text-[#7B8798]">
                      {secondsRemaining > 0
                        ? `OTP expires in ${formattedTime}`
                        : "OTP expired"}
                    </Text>

                    <Pressable
                      onPress={handleSendOtp}
                      disabled={loading || secondsRemaining > 0}
                    >
                      <Text
                        className={`text-sm font-semibold ${
                          secondsRemaining > 0 || loading
                            ? "text-[#A0AAB5]"
                            : "text-[#00BFA5]"
                        }`}
                      >
                        Resend OTP
                      </Text>
                    </Pressable>
                  </View>
                </>
              )}

              {error ? (
                <Text className="mt-4 text-sm text-red-500">{error}</Text>
              ) : null}

              <Pressable
                onPress={otpRequested ? handleLogin : handleSendOtp}
                disabled={loading}
                className={`mt-7 h-14 items-center justify-center rounded-xl ${
                  loading ? "bg-[#80DCCF]" : "bg-[#00BFA5]"
                }`}
              >
                <Text className="text-base font-bold text-white">
                  {loading
                    ? "Please wait..."
                    : otpRequested
                      ? "Verify & Continue"
                      : "Continue"}
                </Text>
              </Pressable>

              {otpRequested && (
                <Pressable
                  onPress={() => {
                    setOtpRequested(false);
                    setOtp("");
                    setSecondsRemaining(0);
                    setError("");
                  }}
                  className="mt-4 items-center"
                >
                  <Text className="text-sm font-semibold text-[#00BFA5]">
                    Change mobile number
                  </Text>
                </Pressable>
              )}
            </View>

            <View className="mt-7 flex-row items-center justify-center">
              <ShieldCheck size={17} color="#00BFA5" />
              <Text className="ml-2 text-xs text-[#7B8798]">
                Your mobile number is safe and secure with us
              </Text>
            </View>
          </View>

import { useEffect, useState } from 'react';
import { useColorScheme as useRNColorScheme } from 'react-native';

/**
 * To support static rendering, this value needs to be re-calculated on the client side for web
 */
export function useColorScheme() {
  const [hasHydrated, setHasHydrated] = useState(false);

  useEffect(() => {
    setHasHydrated(true);
  }, []);

  const colorScheme = useRNColorScheme();

  if (hasHydrated) {
    return colorScheme;
  }

  return 'light';
}

import { io, Socket } from "socket.io-client";
import * as SecureStore from "expo-secure-store";

import { API_BASE_URL } from "@/config/api";

export interface BookingSocketEvent {
  booking: {
    id: string;
    service: string;
    startAt: string;
    status: "confirmed" | "in_progress" | "completed";
    customerName: string;
  };
}

type BookingEventName = "new_booking" | "booking_completed";

type BookingEventHandler = (
  event: BookingSocketEvent,
  eventName: BookingEventName,
) => void;

let shopOwnerSocket: Socket | null = null;
let connectedToken = "";

const listeners = new Set<BookingEventHandler>();

const notifyListeners = (
  event: BookingSocketEvent,
  eventName: BookingEventName,
) => {
  listeners.forEach((listener) => listener(event, eventName));
};

export const connectShopOwnerSocket = async () => {
  const token = await SecureStore.getItemAsync("shopOwnerToken");

  if (!token) {
    disconnectShopOwnerSocket();
    return null;
  }

  if (shopOwnerSocket && connectedToken !== token) {
    disconnectShopOwnerSocket();
  }

  if (!shopOwnerSocket) {
    connectedToken = token;

    shopOwnerSocket = io(API_BASE_URL, {
      auth: { token },
      transports: ["websocket"],
      reconnection: true,
    });

    shopOwnerSocket.on("new_booking", (event: BookingSocketEvent) => {
      notifyListeners(event, "new_booking");
    });

    shopOwnerSocket.on("booking_completed", (event: BookingSocketEvent) => {
      notifyListeners(event, "booking_completed");
    });

    shopOwnerSocket.on("connect_error", (error) => {
      if (
        error.message.includes("no longer authorized") ||
        error.message.includes("Store is inactive")
      ) {
        console.info("[Socket] Store is inactive. Socket disconnected.");
        disconnectShopOwnerSocket();
        return;
      }

      console.error("[Socket] Connection failed:", error.message);
    });
  }

  return shopOwnerSocket;
};

export const subscribeToBookingEvents = (handler: BookingEventHandler) => {
  listeners.add(handler);

  return () => {
    listeners.delete(handler);
  };
};

export const disconnectShopOwnerSocket = () => {
  if (shopOwnerSocket) {
    shopOwnerSocket.removeAllListeners();
    shopOwnerSocket.disconnect();
    shopOwnerSocket = null;
    connectedToken = "";
  }

  listeners.clear();
};

import { useEffect, useRef, useState } from "react";
import * as SecureStore from "expo-secure-store";
import Loader from "@/components/common/Loader";
import Animated, { FadeInRight, FadeOutLeft } from "react-native-reanimated";
import { Image, PanResponder, Pressable, Text, View } from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";
import { useRouter } from "expo-router";
import { ShieldCheck } from "lucide-react-native";
import { API_BASE_URL } from "@/config/api";

const onboardingImages = [
  require("../../assets/images/onboarding/welcome.png"),
  require("../../assets/images/onboarding/discover.png"),
];

export default function HomeScreen() {
  const [currentSlide, setCurrentSlide] = useState(0);
  const router = useRouter();

  const [checkingSession, setCheckingSession] = useState(true);

  useEffect(() => {
    let isMounted = true;

    const checkSession = async () => {
      try {
        const token = await SecureStore.getItemAsync("shopOwnerToken");

        if (token) {
          const response = await fetch(`${API_BASE_URL}/api/shop-owners/me`, {
            headers: {
              Authorization: `Bearer ${token}`,
            },
          });

          const result = await response.json();

          if (response.ok && result.success) {
            router.replace(
              result.data.store.isActive
                ? "/shopowner/dashboard"
                : "/shopowner/profile",
            );
            return;
          }

          await SecureStore.deleteItemAsync("shopOwnerToken");
          router.replace("/auth/login");
          return;
        }
      } catch (error) {
        console.error("[Onboarding] Session check failed.");
      } finally {
        if (isMounted) {
          setCheckingSession(false);
        }
      }
    };

    checkSession();

    return () => {
      isMounted = false;
    };
  }, [router]);

  // 2. Handle swipe gestures between onboarding screens.
  const swipeResponder = useRef(
    PanResponder.create({
      onMoveShouldSetPanResponder: (_, gestureState) =>
        Math.abs(gestureState.dx) > 20 &&
        Math.abs(gestureState.dx) > Math.abs(gestureState.dy),

      onPanResponderRelease: (_, gestureState) => {
        if (gestureState.dx < -60) {
          setCurrentSlide((previous) =>
            Math.min(previous + 1, onboardingImages.length - 1),
          );
        } else if (gestureState.dx > 60) {
          setCurrentSlide((previous) => Math.max(previous - 1, 0));
        }
      },
    }),
  ).current;

  // 1. Handle onboarding navigation.
  const handleNext = () => {
    if (currentSlide < onboardingImages.length - 1) {
      setCurrentSlide((previous) => previous + 1);
      return;
    }

    router.replace("/user/home");
  };

  if (checkingSession) {
    return <Loader fullScreen message="Checking your session..." />;
  }

  return (
    <View className="flex-1 bg-[#EFFFFB]" {...swipeResponder.panHandlers}>
      {/* 2. Display onboarding artwork across the screen. */}
      <Animated.View
        key={currentSlide}
        entering={FadeInRight.duration(450)}
        exiting={FadeOutLeft.duration(300)}
        className="absolute inset-0"
      >
        <Image
          source={onboardingImages[currentSlide]}
          resizeMode="cover"
          className="h-full w-full"
        />
      </Animated.View>

      <SafeAreaView
        className="flex-1 justify-end px-6 pb-24"
        edges={["top", "bottom"]}
      >
        {currentSlide === 1 && (
  <View
    style={{
      position: "absolute",
      top: 45,
      right: 15,
      zIndex: 20,
    }}
  >
    <Pressable
      onPress={() => router.push("/auth/login")}
      className="rounded-full bg-white px-5 py-3"
    >
      <Text className="text-sm font-semibold text-[#00A985]">
        Shop Owner Login
      </Text>
    </Pressable>
  </View>
)}

        {/* 3. Show pagination dots only on the first two screens. */}
        {currentSlide < 2 && (
          <View className="mb-6 flex-row items-center justify-center gap-3">
            {[0, 1].map((index) => (
              <View
                key={index}
                className={`h-3 rounded-full ${
                  currentSlide === index
                    ? "w-7 bg-[#00B894]"
                    : "w-3 bg-[#A7EEE1]"
                }`}
              />
            ))}
          </View>
        )}

        {/* 4. Adjust button spacing separately for each screen. */}
        <View className={currentSlide === 2 ? "mb-28" : ""}>
          <Pressable
            onPress={handleNext}
            className="h-[58px] items-center justify-center rounded-3xl bg-[#00B894] shadow-lg active:opacity-80"
          >
            <Text className="text-lg font-bold text-white">
              {currentSlide === onboardingImages.length - 1
                ? "Get Started"
                : "Next"}
            </Text>
          </Pressable>

          {/* 5. Show terms only on the third screen. */}
          {currentSlide === 2 && (
            <View className="mt-4 flex-row items-center justify-center px-2">
              <ShieldCheck size={24} color="#00B894" />
              <Text className="ml-2 flex-1 text-center text-xs leading-5 text-[#64748B]">
                By continuing, you agree to our{" "}
                <Text className="font-semibold text-[#00A985] underline">
                  Terms and Conditions
                </Text>{" "}
                and{" "}
                <Text className="font-semibold text-[#00A985] underline">
                  Privacy Policy
                </Text>
                .
              </Text>
            </View>
          )}
        </View>
      </SafeAreaView>
    </View>
  );
}
